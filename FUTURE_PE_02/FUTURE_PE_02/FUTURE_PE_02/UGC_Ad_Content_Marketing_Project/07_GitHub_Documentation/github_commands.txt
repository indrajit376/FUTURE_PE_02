# GitHub Upload Commands

# 1. Open terminal inside the project folder
cd UGC_Ad_Content_Marketing_Project

# 2. Initialize Git
git init

# 3. Add all files
git add .

# 4. Commit
git commit -m "Add AI UGC ad content marketing project"

# 5. Rename branch
git branch -M main

# 6. Create a PUBLIC GitHub repository named:
# ai-ugc-ad-content-system

# 7. Connect your repository
git remote add origin YOUR_GITHUB_REPOSITORY_URL

# 8. Push
git push -u origin main
